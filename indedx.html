#include #include #include #include #include #include using namespace std; // ---------- Преобразование числа в слова ---------- string digitToWord(short d, bool female = false) { static const char* male[] = {"ноль","один","два","три","четыре", "пять","шесть","семь","восемь","девять"}; static const char* femaleArr[] = {"ноль","одна","две","три","четыре", "пять","шесть","семь","восемь","девять"}; return female ? femaleArr[d] : male[d]; } string teenToWord(short n) { // 10..19 static const char* words[] = {"десять","одиннадцать","двенадцать","тринадцать", "четырнадцать","пятнадцать","шестнадцать","семнадцать","восемнадцать","девятнадцать"}; return words[n - 10]; } string tensToWord(short n) { // 20..90 кратно 10 static const char* words[] = {"","","двадцать","тридцать","сорок", "пятьдесят","шестьдесят","семьдесят","восемьдесят","девяносто"}; return words[n / 10]; } string hundredsToWord(short n) { static const char* words[] = {"","сто","двести","триста","четыреста", "пятьсот","шестьсот","семьсот","восемьсот","девятьсот"}; return words[n / 100]; } // Триада: число 0..999 -> слова. female — для "одна/две тысячи" string triadToWords(int n, bool female) { string result; short h = n / 100; short t = (n / 10) % 10; short u = n % 10; if (h) result += string(hundredsToWord(h)) + " "; if (t == 1) { result += string(teenToWord(n % 100)) + " "; } else { if (t) result += string(tensToWord(t * 10)) + " "; if (u) result += digitToWord(u, female) + " "; } return result; } // Склонение: 1 рубль / 2 рубля / 5 рублей string pluralForm(long long n, const string& one, const string& few, const string& many) { long long m100 = n % 100; bool negative = kopeks < 0; if (negative) kopeks = -kopeks; long long rub = kopeks / 100; short kop = (short)(kopeks % 100); string result; if (negative) result += "минус "; if (rub == 0) { result += "ноль рублей "; } else { // Разбиваем на триады: млрд, млн, тыс, единицы static const char* triadNames[4] = {"", "тысяч", "миллион", "миллиард"}; static const char* triadNames1[4] = {"", "тысяча", "миллион", "миллиард"}; static const char* triadNames2[4] = {"", "тысячи", "миллиона", "миллиарда"}; static const char* triadNames5[4] = {"", "тысяч", "миллионов", "миллиардов"}; static const bool femaleTriad[4] = {false, true, false, false}; vector triads; long long tmp = rub; while (tmp > 0) { triads.push_back(tmp % 1000); tmp /= 1000; } for (int i = (int)triads.size() - 1; i >= 0; --i) { if (triads[i] == 0) continue; result += triadToWords(triads[i], femaleTriad[i]); if (i > 0) { string form; int t = triads[i]; int m100 = t % 100, m10 = t % 10; if (m100 >= 11 && m100 <= 14) form = triadNames5[i]; else if (m10 == 1) form = triadNames1[i]; else if (m10 >= 2 && m10 <= 4) form = triadNames2[i]; else form = triadNames5[i]; result += form + " "; } } result += pluralForm(rub, "рубль", "рубля", "рублей"); } // Копейки result += " " + to_string(kop) + " "; result += pluralForm(kop, "копейка", "копейки", "копеек"); return result; } // ---------- Демонстрация проблемы float ---------- void demonstrateFloatProblem() { cout << "\n=== Почему float/double плохо для денег ===\n"; float f = 0.1f + 0.2f; double d = 0.1 + 0.2;
